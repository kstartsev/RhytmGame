cmake_minimum_required(VERSION 3.28)
project(WINterArc)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)

include(FetchContent)
FetchContent_Declare(SFML
    GIT_REPOSITORY https://github.com/SFML/SFML.git
    GIT_TAG 3.0.2
    GIT_SHALLOW ON
    EXCLUDE_FROM_ALL
    SYSTEM)

FetchContent_MakeAvailable(SFML)

add_executable(winterarc
    src/main.cpp
    src/core/Game.cpp
    src/entities/Player.cpp
    src/entities/Obstacle.cpp
    src/entities/Borders.cpp
    src/entities/Background.cpp
    src/entities/PercentBar.cpp
    src/entities/Shroom.cpp
    src/core/Context.cpp
    src/core/Scene.cpp
    src/core/EventManager.cpp
    src/core/FileManager.cpp
    src/core/AudioManager.cpp
    src/utils/Utils.cpp
)

target_compile_features(winterarc PRIVATE cxx_std_20)
add_compile_options(-g)
target_link_libraries(winterarc PRIVATE SFML::Graphics SFML::Window SFML::Audio SFML::System)
